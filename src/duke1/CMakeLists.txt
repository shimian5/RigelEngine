cmake_minimum_required(VERSION 3.16)
project(RigelDuke1Sandbox LANGUAGES CXX)
# Standalone entry: does not configure or alter the parent DukeII project.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dn1_core STATIC core.cpp)
target_include_directories(dn1_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(dn1_sandbox sandbox_main.cpp)
target_link_libraries(dn1_sandbox PRIVATE dn1_core)
add_library(dn1_assets STATIC assets.cpp)
target_include_directories(dn1_assets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(dn1_scene scene_main.cpp)
target_link_libraries(dn1_scene PRIVATE dn1_assets)
option(DN1_SANITIZE "Enable ASan/UBSan for standalone validation" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 foreach(t dn1_core dn1_sandbox dn1_assets dn1_scene)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  if(DN1_SANITIZE)
   target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -Werror)
   target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
 endforeach()
endif()
include(CTest)
set(DN1_RESEARCH_ROOT "" CACHE PATH "DOS research folder, test-only (not runtime)")
set(DN1_DATA_DIR "" CACHE PATH "User-supplied original DOS DN1 data, test-only")
if(BUILD_TESTING AND DN1_RESEARCH_ROOT AND DN1_DATA_DIR)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME duke1_core_original_trace_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/validate_core.py" "$<TARGET_FILE:dn1_sandbox>" "${DN1_RESEARCH_ROOT}")
 add_test(NAME duke1_asset_pixel_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/validate_assets.py" "$<TARGET_FILE:dn1_scene>" "${DN1_DATA_DIR}" "${DN1_RESEARCH_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/parity")
 set_tests_properties(duke1_core_original_trace_parity duke1_asset_pixel_parity PROPERTIES TIMEOUT 90)
endif()

add_library(dn1_play STATIC play.cpp actors.cpp)
target_link_libraries(dn1_play PUBLIC dn1_core dn1_assets)
add_executable(dn1_play_shot play_shot_main.cpp)
target_link_libraries(dn1_play_shot PRIVATE dn1_play)
